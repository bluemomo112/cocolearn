## 二、技术选型与约束

### 2.1 GLM-4.7 API 特点

| 特性 | 说明 |
|------|------|
| 接口类型 | OpenAI 兼容的 Chat Completion API |
| 上下文窗口 | 200K tokens |
| 最大输出 | 128K tokens |
| 流式输出 | 支持 |
| Function Call | 支持 |
| Agent API | **不支持**（需通过 Chat 模拟） |

### 2.2 资源文件处理策略

| 文件类型 | 处理方式 |
|----------|----------|
| `.docx` | 提取文本内容，作为上下文传递给 AI |
| `.pptx` | 提取文本和结构，作为上下文传递给 AI |
| `.mp4` | **不传递给 AI**，仅作为用户可查看的资源 |

### 2.3 关键决策

1. **无原生 Agent API**：通过精心设计的 System Prompt + 上下文管理实现多 Agent 效果
2. **资源注入方式**：将文档内容提取后注入到对话上下文中
3. **能力追踪**：AI 在对话中识别能力表现，实时更新前端状态

#### 5.1.2 TypeScript 类型定义

```typescript
// /types/shared-context.ts

interface SharedContext {
  // 1. 会话信息
  session: {
    sessionId: string;
    studentId: string;
    courseId: string;
    startTime: Date;
    currentPhase: 'exploring' | 'learning' | 'practicing';
  };

  // 2. 对话历史 - 所有 Agent 共享
  conversation: {
    messages: Message[];
    lastAgentType: AgentType | null;
    turnCount: number;
    // 标记哪些消息来自哪个 Agent
    messageAgentMap: Map<messageId, AgentType>;
  };

  // 3. 资源上下文
  resources: {
    available: Resource[];
    contents: Map<string, string>;  // resourceId -> extracted text
    currentId: string | null;
    accessLog: Array<{
      resourceId: string;
      startTime: Date;
      duration: number;
    }>;
  };

  // 4. 任务上下文
  tasks: {
    list: Task[];
    status: Map<string, 'locked' | 'available' | 'in_progress' | 'completed'>;
    submissions: Map<string, {
      answer: string;
      submittedAt: Date;
    }>;
    assessments: Map<string, TaskAssessment>;
  };

  // 5. 能力画像
  competency: {
    profile: CompetencyProfile;
    pendingUpdates: CompetencyUpdate[];
    lastUpdated: Date;
  };

  // 6. 行为数据
  behavior: {
    resourceStayTime: Map<string, number>;
    totalIdleTime: number;
    currentIdleStart: Date | null;
    confusionSignals: Array<{
      timestamp: Date;
      signal: string;
      messageId: string;
    }>;
    lastActivityTime: Date;
  };
}

type AgentType = 'tutor' | 'assessor' | 'metacognition';
```

#### 5.1.3 上下文管理 Hook

```typescript
// /hooks/useSharedContext.ts

import { create } from 'zustand';
import { immer } from 'zustand/middleware/immer';

interface SharedContextStore {
  context: SharedContext;

  // ========== 读取方法 ==========
  getConversationHistory: (limit?: number) => Message[];
  getCurrentResource: () => Resource | null;
  getResourceContent: (resourceId: string) => string | null;
  getTaskStatus: (taskId: string) => TaskStatus;
  getCompetencyProfile: () => CompetencyProfile;
  getBehaviorSummary: () => BehaviorSummary;

  // ========== 写入方法 ==========
  addMessage: (message: Message, agentType?: AgentType) => void;
  setCurrentResource: (resourceId: string) => void;
  updateTaskStatus: (taskId: string, status: TaskStatus) => void;
  submitTask: (taskId: string, answer: string) => void;
  addTaskAssessment: (taskId: string, assessment: TaskAssessment) => void;
  updateCompetency: (updates: CompetencyUpdate[]) => void;
  recordBehavior: (event: BehaviorEvent) => void;

  // ========== 会话管理 ==========
  initSession: (studentId: string, courseId: string) => void;
  resetSession: () => void;
}

export const useSharedContext = create<SharedContextStore>()(
  immer((set, get) => ({
    context: createInitialContext(),

    // 获取对话历史（供所有 Agent 使用）
    getConversationHistory: (limit = 20) => {
      const { messages } = get().context.conversation;
      return messages.slice(-limit);
    },

    // 添加消息（标记来源 Agent）
    addMessage: (message, agentType) => {
      set(state => {
        state.context.conversation.messages.push(message);
        state.context.conversation.turnCount++;
        if (agentType) {
          state.context.conversation.lastAgentType = agentType;
          state.context.conversation.messageAgentMap.set(message.id, agentType);
        }
      });
    },

    // 更新能力画像（任务评估器和对话都会调用）
    updateCompetency: (updates) => {
      set(state => {
        updates.forEach(update => {
          const comp = state.context.competency.profile.competencies[update.type];
          if (comp) {
            // 加权更新
            const newRating = (comp.currentRating * comp.evidenceCount + update.rating)
              / (comp.evidenceCount + 1);
            comp.currentRating = newRating;
            comp.evidenceCount++;
            comp.recentEvidence.push(update.evidence);

            // 记录历史
            state.context.competency.profile.history.push({
              timestamp: new Date(),
              competencyType: update.type,
              rating: update.rating,
              source: update.source,
              evidence: update.evidence
            });
          }
        });
        state.context.competency.lastUpdated = new Date();
      });
    },

    // ... 其他方法
  }))
);
```

### 5.2 执行顺序与并发控制

> **核心问题**：多个 Agent 何时触发？是串行还是并行？如何协调？

#### 5.2.1 Agent 触发规则

```
┌─────────────────────────────────────────────────────────────────────────┐
│                        Agent 触发规则                                    │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  用户事件                           触发的 Agent                         │
│  ─────────────────────────────────────────────────────────────────────  │
│                                                                          │
│  1. 发送普通消息                                                         │
│     └──→ Tutor Agent (标准助教)      [串行，阻塞]                        │
│          └──→ Tutor 响应后触发 Metacognition [异步，不阻塞]              │
│               · 分析本轮对话内容，提取能力信号                           │
│               · 输出 → 右侧栏「学习观察」区域                            │
│                                                                          │
│  2. 提交任务答案                                                         │
│     └──→ Assessor Agent (任务评估器) [串行，阻塞]                        │
│          └──→ 评估完成后触发 Metacognition [异步]                        │
│               · 分析任务完成质量，更新能力画像                           │
│               · 输出 → 右侧栏「能力雷达图」+ 「成长记录」                │
│                                                                          │
│  3. 切换/查看资源                                                        │
│     └──→ 仅记录行为日志，不触发 Agent（节省 token）                     │
│          └──→ 行为数据供 Metacognition 下次分析时使用                   │
│                                                                          │
│  ❌ 移除：时间定时触发（每5分钟轮询太耗 token）                          │
│                                                                          │
└─────────────────────────────────────────────────────────────────────────┘

元认知智能体输出位置：右侧栏「学习成长轨迹」区域
├── 能力雷达图（实时更新）
├── AI 观察（每轮对话后更新）
└── 成长记录时间线（任务完成后更新）
```

#### 5.2.2 执行模式：主要串行 + 后台监控并行

```typescript
// /lib/agents/agent-coordinator.ts

type AgentResponse = {
  message: string;
  competencyUpdates?: CompetencyUpdate[];
  shouldTriggerMetacognition?: boolean;
};

class AgentCoordinator {
  private sharedContext: SharedContext;

  constructor(context: SharedContext) {
    this.sharedContext = context;
  }

  /**
   * 处理用户消息 - 主流程（串行）
   */
  async handleUserMessage(userMessage: string): Promise<AgentResponse> {
    // Step 1: 判断消息类型
    const messageType = this.classifyMessage(userMessage);

    // Step 2: 根据类型调用对应 Agent（串行，阻塞）
    let response: AgentResponse;

    switch (messageType) {
      case 'task_submission':
        // 任务提交 → 调用评估器
        response = await this.callAssessorAgent(userMessage);
        // Step 3a: 任务完成后触发元认知分析（异步，不阻塞）
        this.runMetacognitionAnalysis('task_complete', {
          taskId: response.taskId,
          assessment: response.assessment
        });
        break;

      case 'general_question':
      default:
        // 普通问题 → 调用标准助教
        response = await this.callTutorAgent(userMessage);
        // Step 3b: 对话后触发元认知分析（异步，不阻塞）
        this.runMetacognitionAnalysis('chat', {
          userMessage,
          aiResponse: response.message
        });
        break;
    }

    // Step 4: 更新共享上下文
    this.updateContextAfterResponse(response);

    return response;
  }

  /**
   * 调用标准助教 Agent
   */
  private async callTutorAgent(message: string): Promise<AgentResponse> {
    // 构建上下文注入的 System Prompt
    const systemPrompt = buildTutorPrompt({
      resources: this.sharedContext.resources,
      tasks: this.sharedContext.tasks,
      recentConversation: this.sharedContext.conversation.messages.slice(-10),
      competencyProfile: this.sharedContext.competency.profile
    });

    // 调用 GLM
    const response = await chatWithGLM(systemPrompt, [
      ...this.sharedContext.conversation.messages,
      { role: 'user', content: message }
    ]);

    // 解析能力标记
    const competencyUpdates = parseCompetencyMarkers(response);

    return {
      message: response,
      competencyUpdates
    };
  }

  /**
   * 调用任务评估 Agent
   */
  private async callAssessorAgent(submission: string): Promise<AgentResponse> {
    const currentTask = this.getCurrentTask();

    const systemPrompt = buildAssessorPrompt({
      task: currentTask,
      resourceContents: this.sharedContext.resources.contents,
      studentSubmission: submission
    });

    const response = await chatWithGLM(systemPrompt, []);
    const assessment = parseAssessmentResult(response);

    // 更新任务状态
    this.sharedContext.tasks.assessments.set(currentTask.id, assessment);

    // 提取能力更新
    const competencyUpdates = extractCompetencyFromAssessment(assessment);

    return {
      message: formatAssessmentFeedback(assessment),
      competencyUpdates
    };
  }

  /**
   * 元认知分析 - 事件驱动，异步执行
   * 触发时机：Tutor/Assessor 响应后
   * 输出位置：右侧栏「学习成长轨迹」区域
   */
  private async runMetacognitionAnalysis(
    triggerEvent: 'chat' | 'task_complete',
    eventData: any
  ): Promise<void> {
    const systemPrompt = buildMetacognitionPrompt({
      triggerEvent,
      eventData,
      recentConversation: this.sharedContext.conversation.messages.slice(-5),
      behaviorData: this.sharedContext.behavior,
      competencyProfile: this.sharedContext.competency.profile
    });

    const response = await chatWithGLM(systemPrompt, []);
    const result = parseMetacognitionResult(response);

    // ====== 输出到右侧栏，而非对话流 ======

    // 1. 更新能力雷达图数据
    if (result.competencyUpdates?.length > 0) {
      this.sharedContext.competency.pendingUpdates.push(
        ...result.competencyUpdates
      );
      // 触发右侧栏能力雷达图刷新
      this.emitEvent('competency:updated', result.competencyUpdates);
    }

    // 2. 更新「AI 观察」卡片
    if (result.observation) {
      this.emitEvent('observation:new', {
        text: result.observation,
        timestamp: new Date(),
        triggerEvent
      });
    }

    // 3. 添加成长记录时间线条目
    if (triggerEvent === 'task_complete' && result.growthRecord) {
      this.emitEvent('growth:record', result.growthRecord);
    }
  }

  /**
   * 事件发射器 - 通知 UI 更新右侧栏
   */
  private emitEvent(event: string, data: any): void {
    // 使用 Zustand 的订阅机制或自定义事件
    window.dispatchEvent(new CustomEvent(event, { detail: data }));
  }
}
```

#### 5.2.4 Agent 上下文需求定义 (Granular Context Requirements)

> **核心原则**：不同 Agent 需要不同粒度的上下文，避免传递无关数据，节省 token 并提高响应质量。

##### 5.2.4.1 上下文需求对比表

| 上下文数据 | 标准助教 (Tutor) | 任务评估器 (Assessor) | 元认知监控 (Metacognition) |
|-----------|-----------------|---------------------|--------------------------|
| 资源列表 | ✅ 完整列表 | ❌ 不需要 | ✅ 完整列表 |
| 资源查看统计 | ✅ 每个资源的查看次数和总时长 | ❌ 不需要 | ✅ 每个资源的查看次数和总时长 |
| 资源内容摘要 | ✅ 所有资源摘要 | ⚠️ 仅当前任务相关资源 | ❌ 不需要 |
| 任务列表 | ✅ 完整列表 | ❌ 不需要 | ✅ 完整列表 |
| 任务完成状态 | ✅ 所有任务状态 | ❌ 不需要 | ✅ 所有任务状态 |
| 学生答案详情 | ✅ 已完成任务的答案 | ✅ 仅当前任务答案 | ✅ 已完成任务的答案 |
| 当前任务详情 | ❌ 不需要 | ✅ 完整任务配置 | ❌ 不需要 |
| 关联能力维度 | ❌ 不需要 | ✅ 任务指定的能力维度 | ❌ 不需要 |
| 对话历史 | ✅ 最近 10 轮 | ❌ 不需要 | ✅ 最近 5 轮 |
| 能力画像 | ✅ 当前画像 | ❌ 不需要 | ✅ 当前画像 |
| 行为数据 | ❌ 不需要 | ❌ 不需要 | ✅ 完整行为数据 |

```typescript
    },
    recentConversation: sharedContext.conversation.messages.slice(-10),

  };

/**
 * 构建任务评估器上下文（精简版）
 */
function buildAssessorContext(
  sharedContext: SharedContext,
  taskId: string,
  studentAnswer: string
): AssessorAgentContext {
  const task = sharedContext.tasks.list.find(t => t.id === taskId);
  if (!task) throw new Error(`Task ${taskId} not found`);

  // 仅获取与任务相关的资源
  const relevantResourceIds = task.relatedResourceIds || [];
  const relevantResources = relevantResourceIds.map(rid => {
    const resource = sharedContext.resources.available.find(r => r.id === rid);
    const content = sharedContext.resources.contents.get(rid);
    return {
      id: rid,
      title: resource?.title || '',
      // 截取摘要而非全文，节省 token
      contentSummary: content?.slice(0, 1500) || ''
    };
  });

  return {
    currentTask: {
      id: task.id,
      title: task.title,
      type: task.type as 'assignment',
      prompt: task.prompt,
      rubric: task.rubric,
      assignedCompetencies: task.assignedCompetencies || []
    },
    studentSubmission: {
      answer: studentAnswer,
      submittedAt: new Date()
    },
    relevantResources
  };
}

/**
 * 构建元认知监控上下文
 */
function buildMetacognitionContext(
  sharedContext: SharedContext,
  triggerEvent: 'chat' | 'task_complete',
  eventData: any
): MetacognitionAgentContext {
  return {
    triggerEvent,
    eventData,
    resources: {
      list: sharedContext.resources.available.map(r => ({
        id: r.id,
        title: r.title,
        type: r.type
      })),
      // 新增：资源查看统计
      viewStats: buildResourceViewStats(sharedContext)
    },
    tasks: {
      list: sharedContext.tasks.list.map(t => ({
        id: t.id,
        title: t.title,
        type: t.type,
        status: sharedContext.tasks.status.get(t.id) || 'available'
      })),
      // 新增：任务完成详情
      completionDetails: buildTaskCompletionDetails(sharedContext)
    },
    recentConversation: sharedContext.conversation.messages.slice(-5),
    behaviorData: {
      resourceStayTime: sharedContext.behavior.resourceStayTime,
      totalIdleTime: sharedContext.behavior.totalIdleTime,
      confusionSignals: sharedContext.behavior.confusionSignals,
      lastActivityTime: sharedContext.behavior.lastActivityTime
    },
    competencyProfile: sharedContext.competency.profile
  };
}

/**
 * 辅助函数：构建资源查看统计
 */
function buildResourceViewStats(sharedContext: SharedContext): ResourceViewStats[] {
  const { available } = sharedContext.resources;
  const { accessLog } = sharedContext.resources;

  return available.map(resource => {
    // 统计该资源的访问记录
    const resourceLogs = accessLog.filter(log => log.resourceId === resource.id);
    const viewCount = resourceLogs.length;
    const totalViewTime = resourceLogs.reduce((sum, log) => sum + log.duration, 0);
    const lastLog = resourceLogs[resourceLogs.length - 1];

    return {
      resourceId: resource.id,
      title: resource.title,
      viewCount,
      totalViewTime,
      lastViewedAt: lastLog?.startTime || null
    };
  });
}

/**
 * 辅助函数：构建任务完成详情
 */
function buildTaskCompletionDetails(sharedContext: SharedContext): TaskCompletionDetail[] {
  return sharedContext.tasks.list.map(task => {
    const status = sharedContext.tasks.status.get(task.id) || 'available';
    const submission = sharedContext.tasks.submissions.get(task.id);
    const assessment = sharedContext.tasks.assessments.get(task.id);

    return {
      taskId: task.id,
      title: task.title,
      type: task.type,
      status,
      // 仅已提交的任务包含答案
      ...(submission && {
        studentAnswer: submission.answer,
        submittedAt: submission.submittedAt
      }),
      // 仅已评估的任务包含分数
      ...(assessment && {
        score: assessment.score,
        assessment
      })
    };
  });
}
```

```typescript
import OpenAI from 'openai';

const client = new OpenAI({

export async function chatWithGLM(
```

### 6.3 资源内容提取

```typescript
// /lib/resource-parser.ts

import mammoth from 'mammoth';
import { extractTextFromPptx } from './pptx-parser';

export async function extractResourceContent(resource: Resource): Promise<string> {
  switch (resource.fileType) {
    case 'docx':
      const result = await mammoth.extractRawText({ path: resource.path });
      return result.value;

    case 'pptx':
      return await extractTextFromPptx(resource.path);

    case 'mp4':
      // 视频不提取内容
      return `[视频资源: ${resource.title}]`;

    default:
      return '';
  }
}

export async function buildResourceSummary(resources: Resource[]): Promise<string> {
  const summaries = await Promise.all(
    resources.map(async (r) => {
      const content = await extractResourceContent(r);
      // 截取前 2000 字符作为摘要（可根据 token 限制调整）
      const summary = content.slice(0, 2000);
      return `### ${r.title}\n${summary}\n`;
    })
  );

  return summaries.join('\n---\n');
}
```

```typescript
  recentEvidence: string[];
```

### 7.2 实时更新机制

```typescript
// /hooks/useCompetencyTracking.ts

export function useCompetencyTracking() {
  const [profile, setProfile] = useState<CompetencyProfile>(initialProfile);

  // 从 AI 响应中解析能力标记
  const parseCompetencyMarkers = (aiResponse: string): CompetencyUpdate[] => {
    const regex = /<!-- COMPETENCY: (\w+)\|(.+)\|(\d) -->/g;
    const updates: CompetencyUpdate[] = [];

    let match;
    while ((match = regex.exec(aiResponse)) !== null) {
      updates.push({
        type: match[1] as CompetencyType,
        evidence: match[2],
        rating: parseInt(match[3])
      });
    }

    return updates;
  };

  // 更新能力档案
  const updateProfile = (updates: CompetencyUpdate[]) => {
    setProfile(prev => {
      const newProfile = { ...prev };

      updates.forEach(update => {
        const comp = newProfile.competencies[update.type];
        // 加权平均更新评分
        comp.currentRating = (comp.currentRating * comp.evidenceCount + update.rating)
          / (comp.evidenceCount + 1);
        comp.evidenceCount++;
        comp.recentEvidence.push(update.evidence);

        // 更新趋势
        comp.trend = calculateTrend(newProfile.history, update.type);
      });

      return newProfile;
    });
  };

  return { profile, parseCompetencyMarkers, updateProfile };
}
```
